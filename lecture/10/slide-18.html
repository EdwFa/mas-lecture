Голова, которую учим

Путь с новыми весами. MLP читает вектор CLS. У e5 готовых логитов классов нет: вызов с num_labels создаёт случайную голову, пока её не обучили. Обучение головы можно сузить через LoRA.

import torch
from torch import nn
from transformers import AutoModel, AutoTokenizer

name = "intfloat/multilingual-e5-large-instruct"
tok = AutoTokenizer.from_pretrained(name)
enc = AutoModel.from_pretrained(name)

class Head(nn.Module):
    def __init__(self, dim, n):
        super().__init__()
        self.mlp = nn.Sequential(
            nn.Linear(dim, dim), nn.GELU(), nn.Linear(dim, n)
        )
    def forward(self, hidden):
        return self.mlp(hidden)

head = Head(enc.config.hidden_size, 3)

def pack(state, question):
    return f"[STATE] {state} [SEP] [Q] {question}"

@torch.no_grad()
def logits(state, question):
    batch = tok(pack(state, question), return_tensors="pt",
                truncation=True, max_length=512)
    cls = enc(**batch).last_hidden_state[:, 0]
    return head(cls)

Пока голова не обучена, рабочий путь — косинус и softmax на слайдах выше.
