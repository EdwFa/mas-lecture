Проверка и порог

def brier(p, y):
    return float(np.mean((p - y) ** 2))

def ece(p, y, bins=10):
    edges = np.linspace(0, 1, bins + 1)
    idx = np.clip(np.digitize(p, edges) - 1, 0, bins - 1)
    return float(sum(
        (idx == b).mean() * abs(p[idx == b].mean() - y[idx == b].mean())
        for b in range(bins) if (idx == b).any()))

def threshold_for_recall(p, y, target=0.95):
    """Самый высокий порог, при котором полнота >= target."""
    for t in np.sort(np.unique(p))[::-1]:
        if y[p >= t].sum() / y.sum() >= target:
            return float(t)
    return 0.0
