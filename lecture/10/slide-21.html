Голова, которую учим

import torch
from torch import nn
from transformers import AutoModel, AutoTokenizer
name = "intfloat/multilingual-e5-large-instruct"
tok = AutoTokenizer.from_pretrained(name)
enc = AutoModel.from_pretrained(name).eval()
def mean_pool(h, mask):                 # e5 обучена со средним
    m = mask.unsqueeze(-1).float()
    return (h * m).sum(1) / m.sum(1)
head = nn.Sequential(nn.Linear(1024, 256), nn.GELU(),
                     nn.Linear(256, 3))   # 3 варианта choice
opt = torch.optim.AdamW(head.parameters(), lr=1e-3)
def logits(states, questions):
    batch = tok(states, questions, padding=True,
                truncation="only_first", max_length=512,
                return_tensors="pt")      # пара, а не "[SEP]"
    with torch.no_grad():                 # энкодер заморожен
        h = enc(**batch).last_hidden_state
    return head(mean_pool(h, batch["attention_mask"]))
def train_step(states, questions, labels):
    loss = nn.functional.cross_entropy(
        logits(states, questions), labels)
    opt.zero_grad(); loss.backward(); opt.step()
    return loss.item()
