Температура и порог

Температуру подбирают на размеченном наборе: вероятность должна совпасть с частотой верных ответов. Порог сравнивает уже калиброванную вероятность. Оба числа живут в коде. Ориентир набора для своего модуля: 500–1000 фрагментов.

def brier(pairs, temperature):
    loss = 0.0
    for similarity, label in pairs:
        p = float(softmax([similarity, -similarity], temperature)[0])
        loss += (p - label) ** 2
    return loss / len(pairs)

def fit_temperature(pairs):
    best_t, best = 1.0, 1e9
    for t in (0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        loss = brier(pairs, t)
        if loss < best:
            best, best_t = loss, t
    return best_t

def keep(chunks, question, threshold, temperature):
    kept = []
    for chunk in chunks:
        p = yes_probability(question, chunk, temperature)
        if p >= threshold:
            kept.append((chunk, p))
    return kept
