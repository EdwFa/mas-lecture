import re

def check_citations(answer, hits):
    allowed = {c["id"] for c, _ in hits}
    cited = set(re.findall(r"\[([\w./-]+)\]", answer))
    sentences = [s for s in re.split(r"(?<=[.!?])\s+",
                                     answer.strip()) if s]
    return {
        "unknown": sorted(cited - allowed),
        "uncited": [s for s in sentences
                    if not re.search(r"\[[^\]]+\]", s)],
    }
