Лекция 7. Извлечение знаний. RAG20

Код: минимальный RAG

import numpy as np

def build_index(chunks, embed):
    """chunks: [{"id", "text", "source", "version"}]"""
    X = np.array([embed(c["text"]) for c in chunks], dtype=float)
    return X / np.linalg.norm(X, axis=1, keepdims=True)

def search(query, chunks, X, embed, k=4, min_score=0.6):
    q = np.asarray(embed(query), dtype=float)
    scores = X @ (q / np.linalg.norm(q))   # косинус = скалярное
    top = np.argsort(-scores)[:k]
    return [(chunks[i], float(scores[i]))
            for i in top if scores[i] >= min_score]

def build_prompt(question, hits):
    ctx = "\n\n".join(
        f"[{c['id']}] ({c['source']}, {c['version']})\n{c['text']}"
        for c, _ in hits)
    return ("Отвечай только по фрагментам ниже. "
            "После каждой фразы — [id].\n"
            "Если ответа во фрагментах нет — так и скажи.\n\n"
            f"{ctx}\n\nВопрос: {question}")
