Код: цепочка с проверкой

import json, jsonschema

def step(model, prompt, schema, data):
    text = model(prompt.format(
        data=json.dumps(data, ensure_ascii=False)))
    out = json.loads(text)
    jsonschema.validate(out, schema)        # проверка на стыке
    return out

def chain(model, record):
    fields = step(model, EXTRACT, FIELDS, record)   # 1. поля
    issues = check_fields(fields)                   # 2. код
    if issues:
        return {"status": "needs_doctor",
                "issues": issues}
    card = step(model, DRAFT, CARD, fields)         # 3. черновик
    return {"status": "draft", "card": card}

def check_fields(f):
    issues = []
    if f["board_date"] in ("", None):
        issues.append("нет board_date")
    if not f["questions"]:
        issues.append("нет вопросов к консилиуму")
    return issues
