import pathlib, yaml
def scan(root="skills"):
    skills = {}
    for p in pathlib.Path(root).glob("*/SKILL.md"):
        _, front, body = p.read_text(encoding="utf-8").split("---", 2)
        meta = yaml.safe_load(front)
        assert meta["name"] == p.parent.name, p
        skills[meta["name"]] = {"desc": meta["description"],
                                "dir": p.parent, "body": body}
    return skills
def catalog(skills):                    # уровень 1: в системный промпт
    return "\n".join(f"- {n}: {s['desc']}" for n, s in skills.items())
READ_SKILL = {
    "name": "read_skill",
    "description": "Загрузить инструкцию скилла по имени из списка.",
    "input_schema": {"type": "object",
                     "properties": {"name": {"type": "string"}},
                     "required": ["name"],
                     "additionalProperties": False}}
def read_skill(skills, name):           # уровень 2: результат вызова
    s = skills.get(name)
    return s["body"] if s else f"нет скилла {name}"
