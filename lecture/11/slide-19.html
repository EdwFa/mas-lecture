Код: от чанка до точки

import uuid                        # chunks: [{key, text, meta}]
from sentence_transformers import SentenceTransformer
from qdrant_client import QdrantClient, models

model = SentenceTransformer("intfloat/multilingual-e5-large-instruct")
client = QdrantClient(url="http://localhost:6333")

client.create_collection("protocols", vectors_config={
    "dense": models.VectorParams(size=1024,
                                 distance=models.Distance.COSINE)})

vecs = model.encode([c["text"] for c in chunks],      # без префикса
                    normalize_embeddings=True, batch_size=32)
client.upsert("protocols", points=[
    models.PointStruct(
        id=str(uuid.uuid5(uuid.NAMESPACE_URL, c["key"])),
        vector={"dense": v.tolist()},
        payload={**c["meta"], "text": c["text"],
                 "embed_model": "multilingual-e5-large-instruct"})
    for c, v in zip(chunks, vecs)])

q = model.encode(f"Instruct: {TASK}\nQuery: {question}",
                 normalize_embeddings=True)
hits = client.query_points("protocols", query=q.tolist(),
                           using="dense", limit=10).points

Как есть
