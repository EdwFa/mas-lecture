Проверка: recall@k

def recall_at_k(questions, k=10):
    """questions: [(вопрос врача, {id нужных чанков})]"""
    hit = 0
    for question, gold in questions:
        q = model.encode(f"Instruct: {TASK}\nQuery: {question}",
                         normalize_embeddings=True)
        found = {p.id for p in client.query_points(
            "protocols", query=q.tolist(), using="dense",
            limit=k).points}
        hit += bool(found & gold)
    return hit / len(questions)
