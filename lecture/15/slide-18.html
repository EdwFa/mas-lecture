import json, jsonschema

def spawn_subagent(task, contract, tools, model, max_turns=8):
    """Своё окно, свои инструменты, ответ — только по схеме."""
    window = [{"role": "user", "content": brief(task, contract)}]
    for _ in range(max_turns):
        reply = model(window, tools)          # цикл лекции 13
        if reply.tool_calls:                  # права, хуки, трасса —
            window += run_tools(reply.tool_calls, tools)   # как у главного
            continue
        try:
            art = json.loads(reply.text)
            jsonschema.validate(art, contract)
            return art                        # наверх — только артефакт
        except (ValueError, jsonschema.ValidationError) as e:
            window.append({"role": "user",
                           "content": f"Ответ не по схеме: {e}. Верни JSON."})
    return {"status": "failed",
            "reason": f"нет ответа по схеме за {max_turns} оборотов"}

def brief(task, contract):
    return json.dumps({**task, "output_schema": contract},
                      ensure_ascii=False)
