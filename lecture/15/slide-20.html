def stub(name):                       # заглушка вместо модели
    art = json.load(open(f"fixtures/{name}.json"))
    return lambda window, tools: Reply(text=json.dumps(art))

def test_board_date_survives_handoff():
    arts = {n: spawn_subagent(TASKS[n], CONTRACTS[n], [], stub(n))
            for n in READERS}
    pkg = assemble(arts)
    assert pkg["board_date"] is not None, "дата потеряна"
    assert any("две даты" in q for q in pkg["open_questions"])

def test_bad_artifact_is_rejected():
    bad = stub("lab_without_source")
    art = spawn_subagent(TASKS["lab"], CONTRACTS["lab"], [], bad)
    assert art["status"] == "failed"
