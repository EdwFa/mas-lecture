import asyncio

async def search_all(subqueries, search, timeout=5.0):
    """Подзапросы ищем параллельно; медленная ветка
    не держит остальные."""
    missed = []

    async def one(q):
        try:
            return await asyncio.wait_for(search(q), timeout)
        except asyncio.TimeoutError:
            missed.append(q)           # в трассу и в ответ
            return []

    rankings = await asyncio.gather(*(one(s["q"])
                                      for s in subqueries))
    return rrf([r for r in rankings if r]), missed
