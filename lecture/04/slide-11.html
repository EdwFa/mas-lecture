import json, jsonschema

def dispatch(call, tools, impl, policy):
    spec = {t["name"]: t for t in tools}.get(call["name"])
    if spec is None:
        return {"error": f"нет инструмента {call['name']}"}
    try:
        jsonschema.validate(call["args"], spec["input_schema"])
    except jsonschema.ValidationError as e:
        return {"error": f"аргументы: {e.message}"}
    if policy.get(call["name"]) == "ask":
        return {"status": "paused",
                "reason": "нужно подтверждение врача"}
    try:
        return {"result": impl[call["name"]](**call["args"])}
    except TimeoutError:
        return {"error": "МИС не ответила вовремя"}
