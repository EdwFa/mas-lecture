После вызова · маскирование
# после каждого вызова
def mask_phi(result):
    for field in ("name", "address",
                  "telecom", "identifier"):
        result.drop(field)
    result.text = PHI.sub("[скрыто]",
                          result.text)
    return result
Перед ответом · обязательные поля
# перед ответом
REQUIRED = ["case_id", "diagnosis_ref",
            "imaging_ref", "histology_ref",
            "board_date", "open_questions"]

def check_package(pkg):
    missing = [f for f in REQUIRED
               if not pkg.get(f)]
    if missing:
        return Block(f"нет полей: {missing}")

board_date может быть «не назначена». Пустой она быть не может.
