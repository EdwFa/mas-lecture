def test_survives_restart(store, lab, fhir):
    run = start_case("P-0042", store=store, lab=lab, fhir=fhir)
    run.until("wait_lab")
    run.kill()                                 # деплой во вторник

    lab.complete("fixtures/ngs.json")
    lab.push(times=2)                          # уведомление дважды

    run = resume_case("P-0042", store=store, lab=lab, fhir=fhir)
    run.until("paused_for_doctor")
    run.approve(doctor="dr-17")
    run.until("done")

    assert lab.sent_tasks("P-0042") == 1       # задачу не отправили снова
    assert len(fhir.search(identifier=
        "urn:mas:write|P-0042/ngs-result")) == 1   # одна запись
    assert run.trace.has("push.duplicate")
