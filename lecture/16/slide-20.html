def on_push(req, store, seen):
    task_id, stamp = parse_push(req)    # формат — по спецификации
    case = store.case_by_task(task_id)
    if case is None or not hmac.compare_digest(
            req.token, case.push_token):
        return 404                      # одинаково для «нет» и «нельзя»
    event = (task_id, stamp)
    if event in seen:                   # дубль — норма
        return 200
    seen.add(event)
    task = a2a_get_task(task_id)        # тело не принимаем на веру
    store.save(case.with_lab(task))     # сначала сохранить
    schedule(advance, case.id)          # потом шагать
    return 200
