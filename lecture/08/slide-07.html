Код: нарезка по разделам

import re

HEAD = re.compile(r"^(\d+(?:\.\d+)+)\s+\S")
GRADE = re.compile(r"^\(УУР [A-C], УДД [1-5]\)$")

def chunk_guideline(text, doc_id, version, max_chars=1500):
    """Кусок = пункт рекомендаций вместе с оценкой."""
    chunks, section, buf = [], "", []

    def flush():
        if buf:
            chunks.append({"text": " ".join(buf), "section": section,
                           "doc": doc_id, "version": version})
            buf.clear()

    for line in filter(None, map(str.strip, text.splitlines())):
        if GRADE.match(line):          # оценку не отрываем
            buf.append(line)
            continue
        m = HEAD.match(line)
        if m or sum(map(len, buf)) + len(line) > max_chars:
            flush()
            section = m.group(1) if m else section
        buf.append(line)
    flush()
    return chunks
