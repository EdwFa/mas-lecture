def answer(question, case_id, llm, idx):
    subs = plan_queries(question, llm)
    found = []
    for s in subs:
        if s["source"] == "fhir":
            found += fhir_facts(case_id, s["q"])     # лекция 6
        else:
            ids = hybrid_search(s["q"], idx.bm25, idx.dense, idx.meta)
            found += [idx.parent(i) for i in ids]   # малое → большое
    context = select(question, dedup(found), idx.rerank)
    if not context:
        return {"answer": "в рекомендациях не найдено", "sources": []}
    draft = llm(ANSWER.format(question=question, context=context))
    return check_citations(draft, context)           # лекция 7
