def rrf(rankings, k=60):
    """Reciprocal Rank Fusion: сумма 1 / (k + ранг)."""
    score = {}
    for ranking in rankings:
        for rank, doc_id in enumerate(ranking, start=1):
            score[doc_id] = score.get(doc_id, 0) + 1 / (k + rank)
    return sorted(score, key=score.get, reverse=True)

def hybrid_search(query, bm25, dense, meta, top=20):
    """Ищем двумя способами только среди действующих версий."""
    active = {i for i, m in meta.items() if m["status"] == "active"}
    lex = [i for i in bm25(query, top * 2) if i in active]
    sem = [i for i in dense(query, top * 2) if i in active]
    return rrf([lex, sem])[:top]
